"use client";

import { useState } from "react";
import Navbar from "@/components/Navbar";
import Sidebar from "@/components/Sidebar";
import Feed from "@/components/Feed";
import RightPanel from "@/components/RightPanel";
import StoryBar from "@/components/StoryBar";

export default function HomePage() {
  const [activeTab, setActiveTab] = useState<"kesfet" | "akis" | "trend">("akis");

  return (
    <div className="min-h-screen" style={{ background: "#f0f2f5" }}>
      <Navbar />

      {/* Ana Layout */}
      <div
        className="w-full"
        style={{
          marginTop: "60px",
          display: "grid",
          gridTemplateColumns: "240px 1fr 288px",
          gap: "16px",
          alignItems: "start",
          minHeight: "calc(100vh - 60px)",
          padding: "16px 16px 24px",
        }}
      >
        {/* Sol Sidebar */}
        <aside className="min-w-0">
          <div className="sticky" style={{ top: "68px" }}>
            <Sidebar />
          </div>
        </aside>

        {/* Orta Feed */}
        <main className="min-w-0">
          {/* Hikayeler */}
          <StoryBar />

          {/* Tab Bar */}
          <div className="flex bg-white rounded-2xl shadow-sm border border-gray-100 mb-3 overflow-hidden">
            {([
              { key: "akis",   label: "Akış",   icon: "📰" },
              { key: "kesfet", label: "Keşfet", icon: "🔍" },
              { key: "trend",  label: "Trend",  icon: "🔥" },
            ] as const).map((tab, i) => (
              <button
                key={tab.key}
                onClick={() => setActiveTab(tab.key)}
                className="flex-1 flex items-center justify-center gap-2 py-3 font-semibold text-sm transition-all duration-200 relative"

                }
              >
                <span className="text-base leading-none">{tab.icon}</span>
                <span>{tab.label}</span>
                {/* Alt çizgi göstergesi */}
                {activeTab === tab.key && (
                  <span className="absolute bottom-0 left-4 right-4 h-[2.5px] rounded-full" style={{ background: "#e63946" }} />
                )}
                {/* Dikey ayraç */}
                {i < 2 && <span className="absolute right-0 top-3 bottom-3 w-px bg-gray-100" />}
              </button>
            ))}
          </div>

          <Feed activeTab={activeTab} />
        </main>

        {/* Sağ Panel */}
        <aside className="min-w-0">
          <div className="sticky" style={{ top: "68px" }}>
            <RightPanel />
          </div>
        </aside>
      </div>

}
