"use client";
import{useState}from"react";
interface Post{id:number;author:{name:string;username:string;avatar:string;verified:boolean};content:string;image:string|null;likes:number;comments:number;shares:number;time:string;tags:string[];liked:boolean;bookmarked:boolean;}
interface Props{post:Post;onLike:()=>void;onBookmark:()=>void;}
const IG:Record<string,{bg:string;icon:string;label:string}>={gradient1:{bg:"linear-gradient(135deg,#c1121f,#457b9d)",icon:"🚀",label:"Girişim"},gradient2:{bg:"linear-gradient(135deg,#2d6a4f,#52b788)",icon:"🌱",label:"Doğa"},gradient3:{bg:"linear-gradient(135deg,#1d3557,#457b9d)",icon:"💡",label:"Teknoloji"}};
const PAL=[["#e63946","#c1121f"],["#1d3557","#457b9d"],["#2d6a4f","#40916c"],["#e76f51","#f4a261"]];
function grad(n:string){const p=PAL[n.charCodeAt(0)%PAL.length];return `linear-gradient(135deg,${p[0]},${p[1]})`;}
export default function PostCard({post,onLike,onBookmark}:Props){
  const[showC,setShowC]=useState(false),[likeAnim,setLA]=useState(false),[cTxt,setCTxt]=useState(""),[comments,setComments]=useState<string[]>([]),[shared,setShared]=useState(false);
  function doLike(){setLA(true);setTimeout(()=>setLA(false),400);onLike();}
  function doComment(){if(!cTxt.trim())return;setComments(x=>[...x,cTxt]);setCTxt("");}
  function doShare(){setShared(true);setTimeout(()=>setShared(false),1500);}
  const img=post.image?IG[post.image]:null;
  const tc=post.comments+comments.length;
  return(
    <div style={{background:"#fff",borderRadius:16,border:"1px solid #f0f0f0",boxShadow:"0 1px 4px rgba(0,0,0,0.06)",overflow:"hidden"}}>
      <div style={{display:"flex",alignItems:"center",justifyContent:"space-between",padding:"12px 16px 10px"}}>
        <div style={{display:"flex",alignItems:"center",gap:10}}>
          <div style={{width:40,height:40,borderRadius:"50%",background:grad(post.author.name),display:"flex",alignItems:"center",justifyContent:"center",flexShrink:0,boxShadow:"0 2px 4px rgba(0,0,0,0.12)"}}>
            <span style={{color:"#fff",fontWeight:800,fontSize:11}}>{post.author.avatar}</span>
          </div>
          <div>
            <div style={{display:"flex",alignItems:"center",gap:4,marginBottom:2}}>
              <span style={{fontWeight:700,fontSize:13.5,color:"#1c1e21"}}>{post.author.name}</span>
              {post.author.verified&&<span style={{width:14,height:14,background:"#e63946",borderRadius:"50%",display:"inline-flex",alignItems:"center",justifyContent:"center",flexShrink:0}}><svg width="7" height="7" fill="white" viewBox="0 0 24 24"><path d="M20 6 9 17l-5-5"/></svg></span>}
            </div>
            <p style={{fontSize:11,color:"#9ca3af",margin:0}}>@{post.author.username} · {post.time}</p>
          </div>
        </div>
        <button style={{width:28,height:28,border:"none",background:"none",cursor:"pointer",borderRadius:8,display:"flex",alignItems:"center",justifyContent:"center",color:"#9ca3af"}}><svg width="14" height="14" fill="currentColor" viewBox="0 0 24 24"><circle cx="12" cy="5" r="1.5"/><circle cx="12" cy="12" r="1.5"/><circle cx="12" cy="19" r="1.5"/></svg></button>
      </div>
      <div style={{padding:"0 16px 12px"}}>
        <p style={{fontSize:13.5,color:"#1c1e21",lineHeight:1.65,margin:0}}>{post.content}</p>
        {post.tags.length>0&&(<div style={{display:"flex",flexWrap:"wrap",gap:6,marginTop:8}}>{post.tags.map(t=>(<span key={t} style={{fontSize:11,fontWeight:600,color:"#e63946",background:"#fff1f2",border:"1px solid #fecdd3",borderRadius:20,padding:"2px 10px"}}>{t}</span>))}</div>)}
      </div>
      {img&&(<div style={{margin:"0 12px 12px",borderRadius:12,overflow:"hidden",position:"relative",height:130,background:img.bg}}>
        <div style={{position:"absolute",inset:0,background:"linear-gradient(to top,rgba(0,0,0,0.3),transparent)"}}></div>
        <div style={{position:"absolute",inset:0,display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",gap:4}}>
          <span style={{fontSize:32}}>{img.icon}</span><span style={{color:"#fff",fontWeight:600,fontSize:11,opacity:.9}}>{img.label}</span>
        </div></div>)}
      <div style={{display:"flex",alignItems:"center",justifyContent:"space-between",padding:"0 16px 8px"}}>
        <div style={{display:"flex",alignItems:"center",gap:4}}>
          <span style={{width:15,height:15,background:"#e63946",borderRadius:"50%",display:"inline-flex",alignItems:"center",justifyContent:"center"}}><svg width="7" height="7" fill="white" viewBox="0 0 24 24"><path d="M20.84 4.61a5.5 5.5 0 0 0-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 0 0-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 0 0 0-7.78z"/></svg></span>
          <span style={{fontSize:11,color:"#6b7280"}}>{post.likes} beğeni</span>
        </div>
        {tc>0&&<button onClick={()=>setShowC(!showC)} style={{fontSize:11,color:"#9ca3af",background:"none",border:"none",cursor:"pointer",padding:0}}>{tc} yorum</button>}
      </div>
      <div style={{height:1,background:"#f5f5f5"}}></div>
      <div style={{display:"flex",alignItems:"center"}}>
        <button onClick={doLike} style={{flex:1,display:"flex",alignItems:"center",justifyContent:"center",gap:6,padding:"10px 0",border:"none",background:"none",cursor:"pointer",fontSize:12,fontWeight:600,color:post.liked?"#e63946":"#6b7280"}}>
          <svg width="15" height="15" fill={post.liked?"#e63946":"none"} stroke={post.liked?"#e63946":"#6b7280"} strokeWidth="2" viewBox="0 0 24 24" style={{transform:likeAnim?"scale(1.3)":"scale(1)",transition:"transform 0.2s"}}><path d="M20.84 4.61a5.5 5.5 0 0 0-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 0 0-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 0 0 0-7.78z"/></svg>
          {post.liked?"Beğenildi":"Beğen"}
        </button>
        <div style={{width:1,height:16,background:"#f0f0f0"}}></div>
        <button onClick={()=>setShowC(!showC)} style={{flex:1,display:"flex",alignItems:"center",justifyContent:"center",gap:6,padding:"10px 0",border:"none",background:"none",cursor:"pointer",fontSize:12,fontWeight:600,color:"#6b7280"}}>
          <svg width="15" height="15" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24"><path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z"/></svg>Yorum</button>
        <div style={{width:1,height:16,background:"#f0f0f0"}}></div>
        <button onClick={doShare} style={{flex:1,display:"flex",alignItems:"center",justifyContent:"center",gap:6,padding:"10px 0",border:"none",background:"none",cursor:"pointer",fontSize:12,fontWeight:600,color:shared?"#16a34a":"#6b7280"}}>
          <svg width="15" height="15" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24"><path d="M4 12v8a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2v-8"/><polyline points="16 6 12 2 8 6"/><line x1="12" y1="2" x2="12" y2="15"/></svg>{shared?"Paylaşıldı!":"Paylaş"}</button>
        <div style={{width:1,height:16,background:"#f0f0f0"}}></div>
        <button onClick={onBookmark} style={{padding:"10px 14px",border:"none",background:"none",cursor:"pointer",color:post.bookmarked?"#1d3557":"#9ca3af"}}>
          <svg width="15" height="15" fill={post.bookmarked?"#1d3557":"none"} stroke={post.bookmarked?"#1d3557":"currentColor"} strokeWidth="2" viewBox="0 0 24 24"><path d="m19 21-7-5-7 5V5a2 2 0 0 1 2-2h10a2 2 0 0 1 2 2v16z"/></svg>
        </button>
      </div>
      {showC&&(<div style={{borderTop:"1px solid #f5f5f5",padding:"12px 16px",background:"#f9fafb"}}>
        {comments.map((c,i)=>(<div key={i} style={{display:"flex",gap:8,marginBottom:8,alignItems:"flex-start"}}>
          <div style={{width:26,height:26,borderRadius:"50%",background:"linear-gradient(135deg,#e63946,#1d3557)",display:"flex",alignItems:"center",justifyContent:"center",color:"#fff",fontSize:9,fontWeight:700,flexShrink:0}}>AG</div>
          <div style={{background:"#fff",borderRadius:12,padding:"6px 12px",fontSize:12,color:"#1c1e21",flex:1,border:"1px solid #f0f0f0"}}>{c}</div>
        </div>))}
        <div style={{display:"flex",gap:8,alignItems:"center",marginTop:4}}>
          <div style={{width:26,height:26,borderRadius:"50%",background:"linear-gradient(135deg,#e63946,#1d3557)",display:"flex",alignItems:"center",justifyContent:"center",color:"#fff",fontSize:9,fontWeight:700,flexShrink:0}}>AG</div>
          <input value={cTxt} onChange={e=>setCTxt(e.target.value)} onKeyDown={e=>e.key==="Enter"&&doComment()} placeholder="Yorum yaz..." style={{flex:1,border:"1px solid #e5e7eb",borderRadius:20,padding:"7px 14px",fontSize:12,outline:"none",fontFamily:"inherit"}}></input>
          <button onClick={doComment} disabled={!cTxt.trim()} style={{background:"#e63946",color:"#fff",border:"none",borderRadius:20,padding:"7px 14px",fontSize:11,fontWeight:700,cursor:"pointer",opacity:cTxt.trim()?1:0.4,whiteSpace:"nowrap"}}>Gönder</button>
        </div>
      </div>)}
    </div>
  );
}
